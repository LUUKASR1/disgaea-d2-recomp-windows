cmake_minimum_required(VERSION 3.22)
project(D2AndroidBootstrap LANGUAGES C CXX)
option(D2_ANDROID_EMULATOR_SMOKE "CI-only x86_64 Android UI smoke build" OFF)
if(NOT ANDROID OR (NOT ANDROID_ABI STREQUAL "arm64-v8a" AND
        NOT (D2_ANDROID_EMULATOR_SMOKE AND ANDROID_ABI STREQUAL "x86_64")))
    message(FATAL_ERROR "This bootstrap targets Android arm64-v8a only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "Host tests run separately" FORCE)
set(D2_VULKAN_SHADER_COMPILER OFF CACHE BOOL "No desktop DXC in the APK" FORCE)
get_filename_component(D2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
add_subdirectory("${D2_ROOT}/port/vulkan" "${CMAKE_CURRENT_BINARY_DIR}/d2-vulkan")
add_subdirectory("${D2_ROOT}/android/core" "${CMAKE_CURRENT_BINARY_DIR}/d2-core")
add_library(native_app_glue STATIC "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
set_target_properties(native_app_glue PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(native_app_glue PUBLIC "${ANDROID_NDK}/sources/android/native_app_glue")
add_library(d2_android_bootstrap SHARED main.cpp)
target_link_options(d2_android_bootstrap PRIVATE "-uANativeActivity_onCreate" "-Wl,-z,max-page-size=16384")
target_link_libraries(d2_android_bootstrap PRIVATE d2_vulkan d2_android_core native_app_glue android log)
