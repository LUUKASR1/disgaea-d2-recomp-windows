cmake_minimum_required(VERSION 3.24)
project(D2VulkanBase LANGUAGES CXX)
include(CTest)
include(FetchContent)
set(D2_VULKAN_HEADERS_DIR "" CACHE PATH "Existing Khronos Vulkan-Headers checkout")
if(D2_VULKAN_HEADERS_DIR)
    add_subdirectory("${D2_VULKAN_HEADERS_DIR}" "${CMAKE_BINARY_DIR}/headers" EXCLUDE_FROM_ALL)
else()
    FetchContent_Declare(vulkan_headers GIT_REPOSITORY https://github.com/KhronosGroup/Vulkan-Headers.git
        GIT_TAG 29f979ee5aa58b7b005f805ea8df7a855c39ff37)
    FetchContent_MakeAvailable(vulkan_headers)
endif()
add_library(d2_vulkan STATIC vulkan_renderer.cpp)
target_compile_features(d2_vulkan PUBLIC cxx_std_20)
target_include_directories(d2_vulkan PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(d2_vulkan PRIVATE VK_NO_PROTOTYPES NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(d2_vulkan PRIVATE Vulkan::Headers ${CMAKE_DL_LIBS})
add_executable(DisgaeaD2VulkanProbe vulkan_probe.cpp)
target_link_libraries(DisgaeaD2VulkanProbe PRIVATE d2_vulkan)
target_compile_definitions(DisgaeaD2VulkanProbe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(WIN32)
    target_compile_definitions(d2_vulkan PRIVATE VK_USE_PLATFORM_WIN32_KHR)
    target_link_libraries(DisgaeaD2VulkanProbe PRIVATE user32)
    if(MSVC)
        set_property(TARGET DisgaeaD2VulkanProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        set_property(TARGET d2_vulkan PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endif()
if(BUILD_TESTING)
    add_executable(d2_vulkan_geometry_tests tests/geometry_tests.cpp)
    target_link_libraries(d2_vulkan_geometry_tests PRIVATE d2_vulkan)
    add_executable(d2_backend_choice_tests tests/backend_choice_tests.cpp)
    target_compile_features(d2_backend_choice_tests PRIVATE cxx_std_20)
    if(MSVC)
        set_property(TARGET d2_vulkan_geometry_tests d2_backend_choice_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_test(NAME vulkan_geometry_upload COMMAND d2_vulkan_geometry_tests)
    set_tests_properties(vulkan_geometry_upload PROPERTIES TIMEOUT 45 SKIP_RETURN_CODE 77)
    add_test(NAME graphics_backend_selection COMMAND d2_backend_choice_tests)
    add_test(NAME vulkan_offscreen_triangle COMMAND DisgaeaD2VulkanProbe --self-test --report vulkan-test.json)
    set_tests_properties(vulkan_offscreen_triangle PROPERTIES TIMEOUT 45 SKIP_RETURN_CODE 77)
endif()
