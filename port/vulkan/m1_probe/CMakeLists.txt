cmake_minimum_required(VERSION 3.24)
project(D2VulkanM1Probe LANGUAGES CXX)
include(FetchContent)
set(PS3RECOMP_DIR "" CACHE PATH "ps3recomp SDK checkout (headers only are used)")
if(NOT PS3RECOMP_DIR)
    message(FATAL_ERROR "Pass -DPS3RECOMP_DIR=<ps3recomp checkout>")
endif()
set(D2_VULKAN_HEADERS_DIR "" CACHE PATH "Existing Khronos Vulkan-Headers checkout")
if(D2_VULKAN_HEADERS_DIR)
    add_subdirectory("${D2_VULKAN_HEADERS_DIR}" "${CMAKE_BINARY_DIR}/headers" EXCLUDE_FROM_ALL)
else()
    FetchContent_Declare(vulkan_headers
        URL https://github.com/KhronosGroup/Vulkan-Headers/archive/29f979ee5aa58b7b005f805ea8df7a855c39ff37.zip)
    FetchContent_MakeAvailable(vulkan_headers)
endif()
set(V "${CMAKE_CURRENT_LIST_DIR}/..")
add_executable(D2VulkanM1Probe m1_probe.cpp "${V}/rsx_vulkan_backend.cpp" "${V}/vulkan_renderer.cpp")
target_compile_features(D2VulkanM1Probe PRIVATE cxx_std_20)
target_include_directories(D2VulkanM1Probe PRIVATE "${V}"
    "${PS3RECOMP_DIR}/include" "${PS3RECOMP_DIR}/runtime/ppu" "${PS3RECOMP_DIR}/runtime/memory"
    "${PS3RECOMP_DIR}/runtime/platform" "${PS3RECOMP_DIR}/libs/video")
target_compile_definitions(D2VulkanM1Probe PRIVATE VK_NO_PROTOTYPES VK_USE_PLATFORM_WIN32_KHR NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(D2VulkanM1Probe PRIVATE Vulkan::Headers user32)
if(MSVC)
    set_property(TARGET D2VulkanM1Probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
