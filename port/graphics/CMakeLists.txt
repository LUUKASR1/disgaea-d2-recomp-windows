cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(D2Graphics LANGUAGES C CXX)
    include(CTest)
endif()
if(NOT WIN32)
    message(FATAL_ERROR "D2 graphics pipeline requires Windows DirectX 12")
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    generate_shaders.py vendor/fsr1/ffx_a.h vendor/fsr1/ffx_fsr1.h)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_shaders.py"
    "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/fsr_shaders.h" COMMAND_ERROR_IS_FATAL ANY)
add_library(d2_graphics STATIC graphics.cpp capture.cpp hud_capture.cpp hud_present.cpp)
target_compile_features(d2_graphics PUBLIC cxx_std_20)
target_include_directories(d2_graphics PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(d2_graphics PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(d2_graphics PUBLIC d3d12 dxgi d3dcompiler dxguid uuid)
if(BUILD_TESTING)
    add_executable(d2_projection_tests tests/projection_tests.cpp)
    target_link_libraries(d2_projection_tests PRIVATE d2_graphics)
    target_compile_definitions(d2_projection_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_test(NAME world_projection COMMAND d2_projection_tests)
    set_tests_properties(world_projection PROPERTIES TIMEOUT 120)
    add_executable(d2_temporal_probe_tests tests/temporal_probe_tests.cpp)
    target_include_directories(d2_temporal_probe_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(d2_temporal_probe_tests PRIVATE d3dcompiler)
    add_test(NAME temporal_probe COMMAND d2_temporal_probe_tests)
    add_executable(d2_flip_tests tests/flip_tests.cpp)
    target_include_directories(d2_flip_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME flip_pacing COMMAND d2_flip_tests)
    add_executable(d2_graphics_tests tests/graphics_tests.cpp)
    target_link_libraries(d2_graphics_tests PRIVATE d2_graphics)
    target_compile_definitions(d2_graphics_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_test(NAME graphics_gpu COMMAND d2_graphics_tests)
    set_tests_properties(graphics_gpu PROPERTIES TIMEOUT 120)
    add_executable(d2_capture_tests tests/capture_tests.cpp)
    target_link_libraries(d2_capture_tests PRIVATE d2_graphics)
    target_compile_definitions(d2_capture_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_test(NAME capture_control COMMAND d2_capture_tests)
    add_executable(d2_hud_tests tests/hud_tests.cpp)
    target_link_libraries(d2_hud_tests PRIVATE d2_graphics)
    target_compile_definitions(d2_hud_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_test(NAME hud_gpu_replay COMMAND d2_hud_tests)
    set_tests_properties(hud_gpu_replay PROPERTIES TIMEOUT 120)
    add_executable(d2_hud_output_tests tests/hud_output_tests.cpp)
    target_link_libraries(d2_hud_output_tests PRIVATE d2_graphics)
    target_compile_definitions(d2_hud_output_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_test(NAME hud_output_gpu COMMAND d2_hud_output_tests)
    set_tests_properties(hud_output_gpu PROPERTIES TIMEOUT 120)
endif()

set(D2_GRAPHICS_SDK_DIR "" CACHE PATH "Optional audited SDK renderer compile check")
if(D2_GRAPHICS_SDK_DIR)
    set(adapted "${CMAKE_CURRENT_BINARY_DIR}/rsx_d3d12_backend.c")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/adapt_graphics.py"
        "${D2_GRAPHICS_SDK_DIR}/libs/video/rsx_d3d12_backend.c" "${adapted}" COMMAND_ERROR_IS_FATAL ANY)
    set(gcm_adapted "${CMAKE_CURRENT_BINARY_DIR}/cellGcmSys.c")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/adapt_pacing.py"
        "${D2_GRAPHICS_SDK_DIR}/libs/video/cellGcmSys.c" "${gcm_adapted}" COMMAND_ERROR_IS_FATAL ANY)
    add_library(d2_graphics_sdk_check OBJECT "${adapted}" "${gcm_adapted}")
    target_include_directories(d2_graphics_sdk_check PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}"
        "${D2_GRAPHICS_SDK_DIR}/include" "${D2_GRAPHICS_SDK_DIR}/libs/video"
        "${D2_GRAPHICS_SDK_DIR}/libs/system" "${D2_GRAPHICS_SDK_DIR}/runtime/ppu")
endif()
